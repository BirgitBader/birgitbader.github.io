{{- /* Zentrale Ableitung von Titel, Beschreibung und Social-Bild */ -}} {{- /* $title fuer den Browser-Tab und Google, $ogTitle ohne Namenszusatz fuer Social-Karten — dort steht der Name schon in og:site_name. */ -}} {{- $title := "" -}} {{- $ogTitle := "" -}} {{- if .IsHome -}} {{- $title = printf "%s — %s" site.Title site.Params.jobTitle -}} {{- $ogTitle = $title -}} {{- else -}} {{- $title = printf "%s | %s" .Title site.Title -}} {{- $ogTitle = .Title -}} {{- end -}} {{- $description := site.Params.description -}} {{- with .Params.description -}} {{- $description = . -}} {{- else -}} {{- if not .IsHome -}} {{- with .Summary -}} {{- $description = . | plainify | chomp | truncate 160 -}} {{- end -}} {{- end -}} {{- end -}} {{- $ogImage := absURL (.Params.image | default site.Params.ogImage) -}} {{ $title }} {{- /* Nur die echte Domain darf in den Index. Solange die Seite unter birgitbader.github.io (oder lokal) laeuft, bleibt sie auf noindex — sonst entsteht spaeter Duplicate Content zur eigenen Domain. Sobald die Custom Domain in den Pages-Einstellungen hinterlegt ist, faellt das automatisch weg. */ -}} {{- if not (strings.Contains site.BaseURL "birgitbader.com") }} {{- end }} {{- /* Social sharing (LinkedIn, X, Slack, WhatsApp …) */ -}} {{- if and .IsPage (eq .Section "blog") }} {{- with .Lastmod }} {{- end }} {{- end }} {{- /* Icons */ -}} {{- /* Stylesheet mit Cache-Busting-Fingerprint im Dateinamen. Bewusst OHNE integrity="...": Bei einer Datei von der eigenen Domain bringt Subresource Integrity keinen Sicherheitsgewinn — wer das CSS manipulieren koennte, koennte genauso das HTML samt Pruefsumme aendern. Sie schafft aber ein hartes Fehlerbild: Firmen-Proxys und Virenscanner, die Inhalte unterwegs neu komprimieren, brechen die Pruefsumme, und der Browser verwirft das Stylesheet dann komplett. Der Fingerprint im Dateinamen leistet das, worum es hier geht. */ -}} {{- $css := resources.Get "css/style.css" | resources.Minify | resources.Fingerprint "sha256" }} {{- /* RSS-Autodiscovery — auf jeder Seite auf den Site-Feed zeigen */ -}} {{- with site.Home.OutputFormats.Get "rss" }} {{- end }} {{- /* Strukturierte Daten */ -}} {{- if .IsHome }} {{- else if and .IsPage (eq .Section "blog") }} {{- end }}
